package order

// Experimental bounded CDCL on the propositional part of exact one-bit queries.
type shortWatch struct {
	id, blocker int32
}
type shortSolver struct {
	ledger                      *rupLedger
	antecedents                 []int
	valid                       bool
	values                      []int8
	level, reason               []int
	score                       []float64
	saved                       []int8
	clauses                     [][]int32
	watches                     [][]shortWatch
	trail                       []int32
	qhead                       int
	levels                      []int
	seen                        []int
	epoch                       int
	increment                   float64
	steps, conflicts, decisions int
	limit                       int
	mode                        string
}

func shortVar(lit int32) int32 {
	if lit < 0 {
		return -lit
	}
	return lit
}
func shortIndex(lit int32) int {
	if lit < 0 {
		return int(-2*lit + 1)
	}
	return int(2 * lit)
}
func (s *shortSolver) value(lit int32) int8 {
	if lit < 0 {
		return -s.values[-lit]
	}
	return s.values[lit]
}
func (s *shortSolver) put(lit int32, reason int) bool {
	v := shortVar(lit)
	if s.values[v] != 0 {
		return s.value(lit) > 0
	}
	if len(s.levels) == 0 && !s.ledger.rootUnit(reason, lit) {
		s.valid = false
	}
	s.values[v] = 1
	if lit < 0 {
		s.values[v] = -1
	}
	s.level[v] = len(s.levels)
	s.reason[v] = reason
	s.trail = append(s.trail, lit)
	return true
}
func (s *shortSolver) add(lits []int32) int {
	id := len(s.clauses)
	s.clauses = append(s.clauses, lits)
	if len(lits) >= 2 {
		a, b := lits[0], lits[1]
		s.watches[shortIndex(a)] = append(s.watches[shortIndex(a)], shortWatch{int32(id), b})
		s.watches[shortIndex(b)] = append(s.watches[shortIndex(b)], shortWatch{int32(id), a})
	}
	return id
}
func (s *shortSolver) propagate() int {
	for s.qhead < len(s.trail) {
		falseLit := -s.trail[s.qhead]
		s.qhead++
		index := shortIndex(falseLit)
		watches := s.watches[index]
		out := 0
		s.steps += len(watches) + 1
		if s.steps > s.limit {
			return -1
		}
		for pos, w := range watches {
			if s.value(w.blocker) > 0 {
				watches[out] = w
				out++
				continue
			}
			cl := s.clauses[w.id]
			if cl[0] == falseLit {
				cl[0], cl[1] = cl[1], cl[0]
			}
			other := cl[0]
			w.blocker = other
			if s.value(other) > 0 {
				watches[out] = w
				out++
				continue
			}
			moved := false
			for i := 2; i < len(cl); i++ {
				if s.value(cl[i]) < 0 {
					continue
				}
				next := cl[i]
				cl[i], cl[1] = cl[1], next
				idx := shortIndex(next)
				s.watches[idx] = append(s.watches[idx], w)
				moved = true
				break
			}
			if moved {
				continue
			}
			watches[out] = w
			out++
			if !s.put(other, int(w.id)) {
				out += copy(watches[out:], watches[pos+1:])
				s.watches[index] = watches[:out]
				return int(w.id)
			}
		}
		s.watches[index] = watches[:out]
	}
	return 0
}
func (s *shortSolver) analyze(id int) ([]int32, int, bool) {
	s.antecedents = s.antecedents[:0]
	s.epoch++
	stamp := s.epoch
	learned := []int32{0}
	next := len(s.trail) - 1
	level := len(s.levels)
	path := 0
	var pivot int32
	for {
		s.antecedents = append(s.antecedents, id)
		for _, q := range s.clauses[id] {
			v := shortVar(q)
			if v == shortVar(pivot) || s.seen[v] == stamp || s.level[v] == 0 {
				continue
			}
			s.seen[v] = stamp
			s.score[v] += s.increment
			if s.level[v] == level {
				path++
			} else {
				learned = append(learned, q)
			}
		}
		for next >= 0 && s.seen[shortVar(s.trail[next])] != stamp {
			next--
		}
		if next < 0 || path == 0 {
			return nil, 0, false
		}
		pivot = s.trail[next]
		next--
		path--
		if path == 0 {
			break
		}
		id = s.reason[shortVar(pivot)]
		if id == 0 {
			return nil, 0, false
		}
	}
	learned[0] = -pivot
	back := 0
	best := 1
	for i := 1; i < len(learned); i++ {
		lev := s.level[shortVar(learned[i])]
		if lev > back {
			back, best = lev, i
		}
	}
	if len(learned) > 1 {
		learned[1], learned[best] = learned[best], learned[1]
	}
	return learned, back, true
}
func (s *shortSolver) backtrack(target int) {
	if target >= len(s.levels) {
		return
	}
	cut := s.levels[target]
	for _, lit := range s.trail[cut:] {
		v := shortVar(lit)
		s.saved[v] = s.values[v]
		s.values[v] = 0
		s.level[v] = 0
		s.reason[v] = 0
	}
	s.trail = s.trail[:cut]
	s.levels = s.levels[:target]
	s.qhead = cut
}
func (s *shortSolver) choose() int32 {
	v := 0
	score := -1.0
	for i := 1; i < len(s.values); i++ {
		if s.values[i] == 0 && s.score[i] > score {
			v, score = i, s.score[i]
		}
	}
	if v == 0 {
		return 0
	}
	phase := int8(-1)
	if s.mode == "positive" {
		phase = 1
	}
	if s.mode != "positive" && s.saved[v] != 0 {
		phase = s.saved[v]
	}
	return int32(phase) * int32(v)
}

// CompactSolve returns -1 for UNSAT, +1 for SAT, and 0 when a limit was reached.
func (k *StudyCNF) CompactSolve(limit int, mode string) (int, int, int, int) {
	// Expansion caps the number of truth-table and gate variables separately.
	// Clause IDs and signed literal indices fit in the compact watch record.
	if k.Vars < 0 || k.Vars > 200001 || len(k.Clauses) > 1000000 || limit < 1 || int64(limit) > 1<<30 {
		return 0, 0, 0, 0
	}
	n := k.Vars + 1
	s := shortSolver{
		values: make([]int8, n), saved: make([]int8, n), score: make([]float64, n),
		level: make([]int, n), reason: make([]int, n), seen: make([]int, n),
		watches: make([][]shortWatch, 2*n), clauses: make([][]int32, 1, len(k.Clauses)+1),
		limit: limit, mode: mode, increment: 1, valid: true, ledger: newRUPLedger(n, k.Clauses),
	}
	// Reserve the initial two-watch lists together. Each full slice has
	// capacity equal to its own count, so a moved watch cannot overwrite
	// another literal's list when append grows it.
	counts := make([]int, 2*n)
	total := 0
	for ci := range k.Clauses {
		row := k.Clauses[ci].literals()
		for i, lit := range row {
			if lit == 0 || int64(lit) <= -int64(n) || int64(lit) >= int64(n) {
				return 0, 0, 0, 0
			}
			if len(row) >= 2 && i < 2 {
				counts[shortIndex(lit)]++
				total++
			}
		}
	}
	backing := make([]shortWatch, total)
	start := 0
	for i, count := range counts {
		end := start + count
		s.watches[i] = backing[start:start:end]
		start = end
	}
	// The search reorders watches in its own fixed-size copy of each input.
	// Learned clauses can have arbitrary length.
	mutable := make([]shortInputClause, len(k.Clauses))
	for ci := range k.Clauses {
		row := k.Clauses[ci].literals()
		// Check duplicates and tautologies before choosing the first two watches.
		cl := mutable[ci][:0]
		taut := false
		for _, lit := range row {
			repeated := false
			for _, old := range cl {
				if old == lit {
					repeated = true
					break
				}
				if old == -lit {
					taut = true
					break
				}
			}
			if taut {
				break
			}
			if !repeated {
				cl = append(cl, lit)
			}
		}
		if taut {
			// Keep input clause IDs identical in the search and the proof ledger.
			s.add(nil)
			continue
		}
		if len(cl) == 0 {
			if len(row) != 0 {
				return 0, 0, 0, 0
			}
			return -1, 0, 0, 0
		}
		for _, lit := range cl {
			s.score[shortVar(lit)] += 1 / float64(len(cl))
		}
		id := s.add(cl)
		if len(cl) == 1 && !s.put(cl[0], id) {
			if !s.valid || !s.ledger.rootConflict(id) {
				return 0, 0, 0, 0
			}
			return -1, 0, 0, 0
		}
	}
	for {
		conflict := s.propagate()
		if conflict < 0 {
			return 0, s.steps, s.conflicts, s.decisions
		}
		if conflict > 0 {
			s.conflicts++
			if len(s.levels) == 0 {
				if !s.valid || !s.ledger.rootConflict(conflict) {
					return 0, s.steps, s.conflicts, s.decisions
				}
				return -1, s.steps, s.conflicts, s.decisions
			}
			learned, back, ok := s.analyze(conflict)
			if !ok {
				return 0, s.steps, s.conflicts, s.decisions
			}
			if !s.valid || !s.ledger.prove(learned, s.antecedents) {
				return 0, s.steps, s.conflicts, s.decisions
			}
			s.ledger.learned = append(s.ledger.learned, append([]int32(nil), learned...))
			s.backtrack(back)
			id := s.add(learned)
			if !s.put(learned[0], id) {
				return 0, s.steps, s.conflicts, s.decisions
			}
			s.increment *= 1.05
			if s.increment > 1e50 {
				for i := range s.score {
					s.score[i] *= 1e-50
				}
				s.increment *= 1e-50
			}
			if s.mode == "restart" && s.conflicts%80 == 0 {
				s.backtrack(0)
			}
			continue
		}
		next := s.choose()
		if next == 0 {
			// Audit every reported satisfying assignment against the original rows.
			for ci := range k.Clauses {
				row := k.Clauses[ci].literals()
				sat := false
				for _, q := range row {
					if s.value(q) > 0 {
						sat = true
						break
					}
				}
				if !sat {
					return 0, s.steps, s.conflicts, s.decisions
				}
			}
			return 1, s.steps, s.conflicts, s.decisions
		}
		s.decisions++
		s.levels = append(s.levels, len(s.trail))
		s.put(next, 0)
	}
}
