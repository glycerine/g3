package api

import (
	"bufio"
	"encoding/json"
	"os"
	"testing"

	"github.com/glycerine/g3/pkg/util/lbool"
)

func TestWitnessBooleanProofStream(t *testing.T) {
	input := bufio.NewScanner(os.Stdin)
	input.Buffer(nil, 8<<20)
	total, searched := 0, 0
	for input.Scan() {
		var x struct {
			Seed   int
			Script string
		}
		if err := json.Unmarshal(input.Bytes(), &x); err != nil {
			t.Fatal(err)
		}
		func() {
			c, err := NewSMTContext()
			if err != nil {
				t.Fatal(x.Seed, err)
			}
			defer func() {
				if err := c.Close(); err != nil {
					t.Fatal(x.Seed, err)
				}
			}()
			s, err := c.NewSolver()
			if err != nil {
				t.Fatal(x.Seed, err)
			}
			defer func() {
				if err := s.Close(); err != nil {
					t.Fatal(x.Seed, err)
				}
			}()
			if err := s.LoadSMT2String(x.Script); err != nil {
				t.Fatal(x.Seed, err)
			}
			b := s.BeginOrderProofBatch()
			defer func() {
				if err := b.Close(); err != nil {
					t.Fatal(x.Seed, err)
				}
			}()
			if got, err := b.Prove(); err != nil || got {
				t.Fatalf("seed %d model incorrectly refuted %t %v", x.Seed, got, err)
			}
			if err := s.Push(); err != nil {
				t.Fatal(x.Seed, err)
			}
			if err := s.LoadSMT2String(booleanSquare); err != nil {
				t.Fatal(x.Seed, err)
			}
			for i := 0; i < 2; i++ {
				if got, err := b.Prove(); err != nil || !got || b.state.bits == nil {
					t.Fatalf("seed %d missing refutation %d %t %v", x.Seed, i, got, err)
				}
				if b.state.bits.checks > 0 {
					searched++
				}
			}
			if err := s.Pop(1); err != nil {
				t.Fatal(x.Seed, err)
			}
			if got, err := b.Prove(); err != nil || got {
				t.Fatalf("seed %d retained old goal %t %v", x.Seed, got, err)
			}
			if x.Seed%4096 == 0 {
				if ans, err := s.Check(); err != nil || ans != lbool.True {
					t.Fatal(x.Seed, "concrete", ans, err)
				}
				model, err := s.Model()
				if err != nil {
					t.Fatal(x.Seed, err)
				}
				if err := model.Close(); err != nil {
					t.Fatal(x.Seed, err)
				}
			}
		}()
		total++
	}
	if err := input.Err(); err != nil {
		t.Fatal(err)
	}
	t.Logf("verified %d models and %d searched refutations", total, searched)
}
