package order

import "github.com/glycerine/g3/pkg/ast"

// The independent CDCL and proof replay consume the production encoder's CNF.
type StudyCNF BooleanProblem
type shortInputClause = BooleanClause

func CompileOneBit(m *ast.ASTManager, assertions []ast.ASTHandle) (*StudyCNF, bool, error) {
	var t inference
	if ok, err := t.infer(m, assertions); err != nil || !ok {
		return nil, false, err
	}
	k, err := t.booleanProblem(500_000)
	return (*StudyCNF)(k), k != nil, err
}

// PreparedCircuit holds type inference fixed while measuring Boolean encoding.
type PreparedCircuit struct{ terms inference }

func PrepareCircuit(m *ast.ASTManager, assertions []ast.ASTHandle) (*PreparedCircuit, bool, error) {
	var input PreparedCircuit
	ok, err := input.terms.infer(m, assertions)
	if err != nil || !ok {
		return nil, false, err
	}
	return &input, true, nil
}

func (input *PreparedCircuit) Compile() (*BooleanProblem, error) {
	return input.terms.booleanProblem(500_000)
}
