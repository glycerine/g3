package order

import (
	"errors"

	"github.com/glycerine/g3/pkg/ast"
)

// One-bit circuits use signed literals. Literal 1 is constrained true.
type StudyCNF struct {
	Vars      int
	Clauses   []shortInputClause
	nodes     []studyGate
	roots     []studyRoot
	firstGate int32
	work      int
	m         *ast.ASTManager
	gates     map[studyGate]int32
	values    []int32
	tables    [][]int32
}
type studyRoot struct {
	after int
	lit   int32
}
type studyGate struct{ op, a, b, c int32 }

var studyOneBitCap = errors.New("one-bit expansion limit")

// CompileOneBit expands the retained assertions over two-element domains.
// Theory-only conjuncts can be dropped by inference. Only UNSAT establishes
// an answer to the original source problem.
func CompileOneBit(m *ast.ASTManager, assertions []ast.ASTHandle) (*StudyCNF, bool, error) {
	var t inference
	if ok, err := t.infer(m, assertions); err != nil || !ok {
		return nil, false, err
	}
	return t.studyCNF()
}

func (t *inference) studyCNF() (*StudyCNF, bool, error) {
	for _, w := range t.widths {
		if w != 1 {
			return nil, false, nil
		}
	}
	k := &StudyCNF{
		Vars: 1, m: t.m, gates: make(map[studyGate]int32),
		values: make([]int32, len(t.declarations)), tables: make([][]int32, len(t.signatures)),
	}
	k.emit(1, 0, 0)
	for i, s := range t.signatures {
		if len(s.inputs) > 10 || k.Vars+(1<<uint(len(s.inputs))) > 100000 {
			return nil, false, nil
		}
		table := make([]int32, 1<<uint(len(s.inputs)))
		for j := range table {
			k.Vars++
			table[j] = int32(k.Vars)
		}
		k.tables[i] = table
	}
	k.firstGate = int32(k.Vars + 1)
	for _, id := range t.assertions {
		lit, err := k.eval(t, id)
		if err == studyOneBitCap {
			return nil, false, nil
		}
		if err != nil {
			return nil, false, err
		}
		if lit != 1 {
			k.roots = append(k.roots, studyRoot{len(k.nodes), lit})
		}
	}
	k.emitGates()
	return k, true, nil
}
func (k *StudyCNF) tick() error {
	k.work++
	if k.work > 500000 || len(k.nodes) > 100000 || k.work%512 == 0 && k.m.ResourceLimit().CancelFlag() {
		return studyOneBitCap
	}
	return nil
}
func (k *StudyCNF) and(a, b int32) int32 {
	if a == 1 || a == b {
		return b
	}
	if b == 1 {
		return a
	}
	if a == -1 || b == -1 || a == -b {
		return -1
	}
	if a > b {
		a, b = b, a
	}
	key := studyGate{op: 1, a: a, b: b}
	if n := k.gates[key]; n != 0 {
		return n
	}
	k.Vars++
	x := int32(k.Vars)
	k.gates[key] = x
	k.nodes = append(k.nodes, key)
	return x
}
func (k *StudyCNF) ite(c, a, b int32) int32 {
	if c == 1 || a == b {
		return a
	}
	if c == -1 {
		return b
	}
	if a == 1 {
		return -k.and(-c, -b)
	}
	if a == -1 {
		return k.and(-c, b)
	}
	if b == 1 {
		return -k.and(c, -a)
	}
	if b == -1 {
		return k.and(c, a)
	}
	if c < 0 {
		a, b = b, a
		c = -c
	}
	neg := int32(1)
	if a < 0 {
		a, b = -a, -b
		neg = -1
	}
	key := studyGate{op: 2, a: c, b: a, c: b}
	if n := k.gates[key]; n != 0 {
		return neg * n
	}
	k.Vars++
	x := int32(k.Vars)
	k.gates[key] = x
	k.nodes = append(k.nodes, key)
	return neg * x
}

// Select one entry of an uninterpreted function's complete truth table.
func (k *StudyCNF) lookup(args []int32, table []int32) int32 {
	if len(args) == 0 {
		return table[0]
	}
	half := len(table) / 2
	sel := args[0]
	if sel == 1 {
		return k.lookup(args[1:], table[half:])
	}
	if sel == -1 {
		return k.lookup(args[1:], table[:half])
	}
	return k.ite(sel, k.lookup(args[1:], table[half:]), k.lookup(args[1:], table[:half]))
}
func (k *StudyCNF) eval(t *inference, id int32) (int32, error) {
	if e := k.tick(); e != nil {
		return 0, e
	}
	n := &t.terms[id]
	switch n.op {
	case opTrue:
		return 1, nil
	case opFalse:
		return -1, nil
	case opVariable:
		return k.values[n.value], nil
	case opNumeral:
		if n.value == 1 {
			return 1, nil
		}
		return -1, nil
	case opForall, opExists:
		binders := t.declarations[n.start : n.start+n.count]
		if len(binders) > 12 {
			return 0, studyOneBitCap
		}
		all := n.op == opForall
		res := int32(1)
		for mask := 0; mask < 1<<uint(len(binders)); mask++ {
			for j, b := range binders {
				v := int32(-1)
				if mask&(1<<uint(j)) != 0 {
					v = 1
				}
				k.values[b.name] = v
			}
			r, e := k.eval(t, n.value)
			if e != nil {
				return 0, e
			}
			if !all {
				r = -r
			}
			res = k.and(res, r)
			if res == -1 {
				break
			}
		}
		if !all {
			res = -res
		}
		return res, nil
	}
	var small [8]int32
	args := small[:0]
	for _, a := range t.arguments(n) {
		v, e := k.eval(t, a)
		if e != nil {
			return 0, e
		}
		args = append(args, v)
	}
	switch n.op {
	case opFunction:
		return k.lookup(args, k.tables[n.value]), nil
	case opNot:
		return -args[0], nil
	case opAnd, opOr:
		or := n.op == opOr
		r := int32(1)
		for _, a := range args {
			if or {
				a = -a
			}
			r = k.and(r, a)
		}
		if or {
			r = -r
		}
		return r, nil
	case opXor:
		return k.ite(args[0], -args[1], args[1]), nil
	case opEqual, opDistinct:
		if n.op == opDistinct && len(args) > 2 {
			return -1, nil
		}
		r := int32(1)
		for _, a := range args[1:] {
			r = k.and(r, k.ite(args[0], a, -a))
		}
		if n.op == opDistinct {
			r = -r
		}
		return r, nil
	case opIte:
		return k.ite(args[0], args[1], args[2]), nil
	case opImplies:
		return -k.and(args[0], -args[1]), nil
	case opOrder:
		a, b := args[0], args[1]
		switch n.value {
		case bvULT:
			return k.and(-a, b), nil
		case bvUGT:
			return k.and(a, -b), nil
		case bvULE:
			return -k.and(a, -b), nil
		case bvUGE:
			return -k.and(-a, b), nil
		}
	}
	return 0, inapplicable
}

// A positive occurrence of a gate needs only the forward implication.
// Mark dependencies in reverse construction order before writing clauses.
func (k *StudyCNF) emitGates() {
	need := make([]uint8, k.Vars+1)
	mark := func(x int32) {
		if x < 0 {
			need[-x] |= 2
		} else {
			need[x] |= 1
		}
	}
	for _, root := range k.roots {
		mark(root.lit)
	}
	for i := len(k.nodes) - 1; i >= 0; i-- {
		g := k.nodes[i]
		x := k.firstGate + int32(i)
		flags := need[x]
		if flags == 0 {
			continue
		}
		a, b := g.a, g.b
		if g.op == 2 {
			mark(a)
			mark(-a)
			a, b = g.b, g.c
		}
		if flags&1 != 0 {
			mark(a)
			mark(b)
		}
		if flags&2 != 0 {
			mark(-a)
			mark(-b)
		}
	}
	next := 0
	for i := 0; i <= len(k.nodes); i++ {
		for next < len(k.roots) && k.roots[next].after == i {
			lit := k.roots[next].lit
			k.emit(lit, 0, 0)
			next++
		}
		if i == len(k.nodes) {
			break
		}
		g := k.nodes[i]
		x := k.firstGate + int32(i)
		flags := need[x]
		if g.op == 1 {
			if flags&1 != 0 {
				k.emit(-x, g.a, 0)
				k.emit(-x, g.b, 0)
			}
			if flags&2 != 0 {
				k.emit(x, -g.a, -g.b)
			}
		} else {
			if flags&1 != 0 {
				k.emit(-g.a, -x, g.b)
			}
			if flags&2 != 0 {
				k.emit(-g.a, x, -g.b)
			}
			if flags&1 != 0 {
				k.emit(g.a, -x, g.c)
			}
			if flags&2 != 0 {
				k.emit(g.a, x, -g.c)
			}
		}
	}
}

type shortInputClause [3]int32

func (c *shortInputClause) literals() []int32 {
	if c[2] != 0 {
		return c[:3]
	}
	if c[1] != 0 {
		return c[:2]
	}
	if c[0] != 0 {
		return c[:1]
	}
	return nil
}
func (k *StudyCNF) emit(a, b, c int32) { k.Clauses = append(k.Clauses, shortInputClause{a, b, c}) }
