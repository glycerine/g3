package order

import (
	"math/rand"
	"testing"
)

func TestStudyCompactTruthTable(t *testing.T) {
	r := rand.New(rand.NewSource(617))
	cases := 0
	for i := 0; i < 70000; i++ {
		n := 1 + r.Intn(10)
		k := StudyCNF{Vars: n}
		rows := r.Intn(13) + 4*n
		for j := 0; j < rows; j++ {
			l := 1 + r.Intn(3)
			if i%157 == 0 {
				l = 0
			}
			cl := make([]int32, l)
			for a := range cl {
				cl[a] = int32(1 + r.Intn(n))
				if r.Intn(2) == 1 {
					cl[a] = -cl[a]
				}
			}
			var row shortInputClause
			copy(row[:], cl)
			k.Clauses = append(k.Clauses, row)
		}
		want := -1
		for bits := 0; bits < 1<<uint(n); bits++ {
			model := true
			for _, cl := range k.Clauses {
				sat := false
				for _, lit := range cl {
					if lit == 0 {
						break
					}
					truth := bits&(1<<uint(shortVar(lit)-1)) != 0
					if truth == (lit > 0) {
						sat = true
						break
					}
				}
				if !sat {
					model = false
					break
				}
			}
			if model {
				want = 1
				break
			}
		}
		for _, mode := range []string{"phase", "positive", "restart"} {
			got, _, _, _ := k.CompactSolve(5000000, mode)
			if got != want {
				t.Fatalf("case %d, mode %s, got %d, want %d, CNF %v", i, mode, got, want, k.Clauses)
			}
			cases++
		}
		reduced := StudyCNF{Vars: n, Clauses: append([]shortInputClause(nil), k.Clauses...)}
		got := -1
		if !reduced.StudyUnits(16) {
			got, _, _, _ = reduced.CompactSolve(5000000, "positive")
		}
		if got != want {
			t.Fatalf("case %d after unit propagation, got %d, want %d", i, got, want)
		}
	}
	t.Logf("%d exhaustive comparisons", cases)
}

func TestStudyCompactDeclinesInvalidLiteral(t *testing.T) {
	for _, literal := range []int32{2, -2, -1 << 31} {
		k := StudyCNF{Vars: 1, Clauses: []shortInputClause{{literal}}}
		if got, _, _, _ := k.CompactSolve(100, "positive"); got != 0 {
			t.Fatalf("literal %d produced answer %d", literal, got)
		}
	}
}

func TestStudyRefutationRejectsUnprovedClauses(t *testing.T) {
	l := newRUPLedger(5)
	l.addOriginal([]int32{1, 2})
	l.addOriginal([]int32{-1, 3})
	l.addOriginal([]int32{-2, 3})
	if !l.prove([]int32{3}, []int{3, 2, 1}) && !l.prove([]int32{3}, []int{1, 2, 3}) {
		t.Fatal("valid resolution missing")
	}
	if l.prove([]int32{-3}, []int{3, 2, 1}) || l.prove([]int32{3}, []int{2, 1}) || l.rootUnit(2, 3) || l.rootConflict(1) {
		t.Fatal("invalid reason accepted")
	}
}
