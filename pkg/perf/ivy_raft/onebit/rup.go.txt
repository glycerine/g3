package order

// A checked clause follows by reverse unit propagation from previously
// accepted clauses. The search supplies the antecedents in conflict order.
type rupLedger struct {
	original []shortInputClause
	learned  [][]int32
	root     []int8
	mark     []uint32
	value    []int8
	stamp    uint32
}

func newRUPLedger(n int, original []shortInputClause) *rupLedger {
	return &rupLedger{original: original, root: make([]int8, n), mark: make([]uint32, n), value: make([]int8, n)}
}

func (p *rupLedger) begin() {
	p.stamp++
	if p.stamp == 0 {
		clear(p.mark)
		p.stamp = 1
	}
}
func (p *rupLedger) read(lit int32) int8 {
	i := lit
	if i < 0 {
		i = -i
	}
	v := p.root[i]
	if v == 0 && p.mark[i] == p.stamp {
		v = p.value[i]
	}
	if lit < 0 {
		return -v
	}
	return v
}
func (p *rupLedger) assign(lit int32) {
	i := lit
	v := int8(1)
	if i < 0 {
		i = -i
		v = -1
	}
	p.mark[i] = p.stamp
	p.value[i] = v
}

// A falsified row returns conflict. A unit returns its only possible literal.
// Rows that are already true or have several possible literals imply nothing.
func (p *rupLedger) scan(id int) (unit int32, conflict bool) {
	if id <= 0 {
		return 0, false
	}
	var row []int32
	if id <= len(p.original) {
		row = p.original[id-1].literals()
	} else {
		i := id - len(p.original) - 1
		if i >= len(p.learned) {
			return 0, false
		}
		row = p.learned[i]
	}
	for _, lit := range row {
		switch p.read(lit) {
		case 1:
			return 0, false
		case 0:
			if unit != 0 && unit != lit {
				return 0, false
			}
			unit = lit
		}
	}
	return unit, unit == 0
}
func (p *rupLedger) rootUnit(id int, lit int32) bool {
	p.begin()
	unit, _ := p.scan(id)
	if lit == 0 || unit != lit {
		return false
	}
	if lit < 0 {
		p.root[-lit] = -1
	} else {
		p.root[lit] = 1
	}
	return true
}
func (p *rupLedger) rootConflict(id int) bool {
	p.begin()
	_, conflict := p.scan(id)
	return conflict
}
func (p *rupLedger) prove(clause []int32, reasons []int) bool {
	p.begin()
	for _, q := range clause {
		if q == 0 || int64(q) <= -int64(len(p.root)) || int64(q) >= int64(len(p.root)) {
			return false
		}
		switch p.read(q) {
		case 1:
			return true
		case 0:
			p.assign(-q)
		}
	}
	for j := len(reasons) - 1; j >= 0; j-- {
		unit, conflict := p.scan(reasons[j])
		if conflict {
			return true
		}
		if unit != 0 {
			p.assign(unit)
		}
	}
	return false
}
