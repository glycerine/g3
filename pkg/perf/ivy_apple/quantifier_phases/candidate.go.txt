// Linux/amd64, linked only into the single-worker measurement binary.
package smtcore

import (
	"fmt"
	"os"
	"syscall"
	"unsafe"
)

var qiObserved *qiCandidateCounter
var qiCounterFD = -1
var qiSequence int

type qiCandidateCounter struct {
	previous                 *qiCandidateCounter
	ordinal, bindings, phase int
	id, generation           uint32
	last, empty              uint64
	instructions             [7]uint64
	outcome                  string
}

func qiReadCount() uint64 {
	var result [3]uint64
	size, _, err := syscall.RawSyscall(syscall.SYS_READ, uintptr(qiCounterFD), uintptr(unsafe.Pointer(&result)), unsafe.Sizeof(result))
	if err != 0 || size != unsafe.Sizeof(result) || result[1] != result[2] {
		panic("invalid QI thread count")
	}
	return result[0]
}

func qiStartCounter(id, gen uint32, bindings int) {
	if qiCounterFD < 0 {
		var attr struct {
			Type, Size                            uint32
			Config, Period, Sample, Format, Flags uint64
			Reserved                              [16]byte
		}
		attr.Size = uint32(unsafe.Sizeof(attr))
		attr.Config, attr.Format, attr.Flags = 1, 3, 1<<5|1<<6
		fd, _, err := syscall.RawSyscall6(syscall.SYS_PERF_EVENT_OPEN, uintptr(unsafe.Pointer(&attr)), 0, ^uintptr(0), ^uintptr(0), 0, 0)
		if err != 0 {
			panic(err)
		}
		qiCounterFD = int(fd)
	}
	qiSequence++
	c := &qiCandidateCounter{previous: qiObserved, ordinal: qiSequence, id: id, generation: gen, bindings: bindings, outcome: "satisfied"}
	qiObserved = c
	a := qiReadCount()
	c.last = qiReadCount()
	c.empty = c.last - a
}

func qiNextPhase(next int) {
	c := qiObserved
	if c == nil {
		return
	}
	now := qiReadCount()
	delta := now - c.last
	if delta > c.empty {
		c.instructions[c.phase] += delta - c.empty
	}
	c.last, c.phase = now, next
}

func qiEndCounter() {
	c := qiObserved
	if c == nil {
		return
	}
	qiNextPhase(6)
	qiObserved = c.previous
	fmt.Fprintln(os.Stderr, "IVY-QI", c.ordinal, c.id, c.generation, c.bindings, c.outcome, c.empty,
		c.instructions[0], c.instructions[1], c.instructions[2], c.instructions[3], c.instructions[4], c.instructions[5], c.instructions[6])
}

func qiStartEntry(q *SMTQuantifierManager, entry SMTQIQueueEntry) {
	if os.Getenv("IVY_QI_ACTIVE") == "" {
		return
	}
	id, err := q.state.manager.GetASTID(entry.Fingerprint.data)
	if err != nil {
		panic(err)
	}
	qiStartCounter(id, entry.Generation, len(entry.Fingerprint.args))
}
