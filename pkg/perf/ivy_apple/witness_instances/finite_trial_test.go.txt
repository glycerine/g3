package api

import (
	"bytes"
	"fmt"
	"os"
	"testing"

	"github.com/glycerine/g3/pkg/ast"
	"github.com/glycerine/g3/pkg/solver"
)

// finite_trials.py builds this test through a Go overlay. Each process receives
// one SMT2 input with an independently checked finite model.
func TestKnownFiniteWitnessModel(t *testing.T) {
	raw, err := os.ReadFile(os.Getenv("G3_WITNESS_MODEL"))
	astTestMust(t, err)
	raw, _, _ = bytes.Cut(raw, []byte("\n(check-sat)"))
	c, s := orderSolver(t, string(raw))
	m, err := c.Manager()
	astTestMust(t, err)
	forms, err := solver.Assertions(s.solver)
	astTestMust(t, err)
	normal, err := planWitnessInstances(m, forms)
	astTestMust(t, err)
	if yes, err := s.ProveUnsatByOrderAbstraction(); yes || err != nil {
		t.Fatalf("public proof on a satisfiable input returned %v, %v", yes, err)
	}

	var domains []ast.ASTHandle
	sorts := make(map[ast.ASTHandle]bool)
	seen := make(map[ast.ASTHandle]bool)
	stack := append([]ast.ASTHandle(nil), forms...)
	for len(stack) != 0 {
		h := stack[len(stack)-1]
		stack = stack[:len(stack)-1]
		if seen[h] {
			continue
		}
		seen[h] = true
		head, err := m.NodeHeader(h)
		astTestMust(t, err)
		switch head.Kind {
		case ast.ASTQuantifier:
			n, err := m.GetNumDeclarations(h)
			astTestMust(t, err)
			for i := uint32(0); i < n; i++ {
				st, err := m.GetDeclarationSort(h, i)
				astTestMust(t, err)
				u, err := m.IsUninterpretedSort(st)
				astTestMust(t, err)
				if u && !sorts[st] {
					sorts[st] = true
					domains = append(domains, st)
				}
			}
			body, err := m.GetQuantifierBody(h)
			astTestMust(t, err)
			stack = append(stack, body)
		case ast.ASTApp:
			for i := uint32(0); i < head.NumArgs; i++ {
				arg, err := m.GetArg(h, i)
				astTestMust(t, err)
				stack = append(stack, arg)
			}
		}
	}
	if len(domains) == 0 {
		t.Fatal("expected a quantified uninterpreted sort")
	}
	p := orderProofState{source: s}
	trial := &witnessInstancePlan{assertions: forms, sorts: domains, width: 4, rounds: 2}
	yes, err := p.checkProofPlan(m, trial, 1_500_000)
	if yes || err != nil {
		t.Fatalf("forced proof on a satisfiable input returned %v, %v", yes, err)
	}
	fmt.Println("FINITE", len(forms), len(domains), normal != nil, yes)
}
